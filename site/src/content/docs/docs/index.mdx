import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

DVM creates and starts [Lima](https://lima-vm.io) instances, keeps host files
out of the guest, creates a dev user, and runs your own setup scripts. Tool
installation lives in those scripts.

<CardGrid>
  <Card title="No host mounts" icon="laptop">
    Nothing from your machine is mounted into the guest by default.
  </Card>
  <Card title="Your setup, your scripts" icon="seti:shell">
    Provisioning is plain Bash you own and review.
  </Card>
  <Card title="Checked before they run" icon="approve-check">
    Setup scripts are ownership- and permission-checked first.
  </Card>
  <Card title="Disposable by design" icon="trash">
    Spin up, build, throw away. The host stays clean.
  </Card>
</CardGrid>
